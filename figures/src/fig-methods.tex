\newcommand{\methodrow}[8]{%
  \node[rowlab, text=#4!55!black] at (\xlab, #2) {#3};
  \node[io] (i#1) at (\xin, #2) {segment};
  \node[bb, draw=#4!68!black, fill=#4!16] (b#1) at (\xbb, #2) {#5};
  \node[io] (o#1) at (\xout, #2) {#7};
  \draw[edge] (i#1) -- (b#1);
  \ifthenelse{\equal{#6}{}}{%
    \draw[edge] (b#1) -- (o#1);
  }{%
    \node[lab, draw=#4!60!black, fill=#4!11] (h#1) at (\xhd, #2) {#6};
    \draw[edge] (b#1) -- (h#1);
    \draw[edge] (h#1) -- (o#1);
  }%
  \node[note, anchor=west, align=left, text width=3.3cm] at (\xnote, #2) {#8};
}
